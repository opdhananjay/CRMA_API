CREATE PROCEDURE [dbo].[sp_get_user_by_email] (
	@Email NVARCHAR(50)
)
AS
BEGIN 
	
	SET NOCOUNT ON;

	SELECT UserId,FullName,Email,PasswordHash,PasswordSalt,IsActive,PlatformRole FROM dbo.Users
	WHERE Email = @Email

END
